def one_to_one_matching(self, detections):
        unassigned = detections[(detections.tracker_id == -1) | (detections.tracker_id > 5)]
        canon = np.array(range(0,6)) 
        keep_ids = ~np.isin(canon, detections.tracker_id)
        unused_canon = []
        unused_canon = canon[keep_ids]
        
        #print(f"unused canon : {unused_canon}")
        #print(f"unassigned : {unassigned}")
        #check unassigned and unused canon len
        if len(unused_canon)==1 and len(unassigned)== 1:
            #print(detections.xyxy)
            #print(f"unassigned xyxy {unassigned.xyxy}")
            mask = np.all(detections.xyxy == unassigned.xyxy, axis = 1)
            detections.tracker_id[mask] = unused_canon[0]
            #print(unassigned)
            #print(detections[detections == unassigned].tracker_id)
        #print(detections.tracker_id)
        return detections, unassigned, unused_canon
